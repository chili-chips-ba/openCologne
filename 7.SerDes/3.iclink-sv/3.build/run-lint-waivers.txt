# Accepted lint findings, read by run-lint.sh.
#
# One waiver per line:  <phase> <regex>
#   <phase>  tags | verilator | yosys | slang | verible
#   <regex>  extended regex (grep -E), matched against one finding line as it
#            appears in <phase>.all of a run-lint.sh work directory
# A finding that matches is waived. Put the reason on a comment line above it.
#
# Guideline: resolve a finding at its root; waive only as the last resort, by
# the same steps as openPCIE's run-lint-waivers.txt:
#   1. dead logic in our RTL: delete it;
#   2. oversized buses in our RTL: size them to what is used;
#   3. unused outputs of hard macros, primitives and third-party cores:
#      .PIN (/*LINT:OK*/), or route them to an NC_ (no-connect) signal;
#   4. only then a waiver here, with its reason.

# a package serves several designs (top, the SerWB link layer): a constant
# that one of them does not use is not dead
verilator ^%Warning-UNUSEDPARAM: 1\.hw/iclink_[a-z_]*pkg\.sv:

# the SerWB modules size their casts for any W; at W = 2 some are 1 bit already
slang ^1\.hw/iclink_serwb_(fifo|rx)\.sv:[0-9]+:[0-9]+: warning: useless cast

# a unique case that lists only the legal values of a wider selector (CRC
# term counts 0 .. 4, the three line rates) ends in an empty default, as our
# rules ask; slang calls that default redundant, Verilator wants it
slang ^1\.hw/iclink_(serwb|phy)_pkg\.sv:[0-9]+:[0-9]+: warning: 'case' marked 'unique' has 'default' label

# NC_ signals are no-connects by name: driven by an unused output, never read
verilator ^%Warning-UNUSEDSIGNAL: .*'NC_[A-Za-z0-9_]+'

# iclink_serwb_line at W = 2 (its default, as lint takes it on its own) is a
# pass-through with no flop: clk and arst_n run only the W = 4 conversion
verilator ^%Warning-UNUSEDSIGNAL: 1\.hw/iclink_serwb_line\.sv:[0-9]+:[0-9]+: Signal is not used: '(clk|arst_n)'
